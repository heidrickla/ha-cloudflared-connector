#include <tunables/global>

profile cloudflared_connector flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>
  #include <abstractions/ssl_certs>

  # Files: the rule set the Home Assistant example profiles use for an
  # s6-overlay based image, so init and the connector can run.
  file,
  signal (send) set=(kill,term,int,hup,cont),
  signal (receive) set=(kill,term,int,hup,cont),

  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /tmp/** rwk,

  # add-on options and state
  /data/** rw,

  # The connector, a static Rust binary in this profile: it starts and signals
  # cloudflared and serves the panel to the ingress proxy over inet stream.
  /usr/local/bin/connector rmix,

  # cloudflared runs in this profile: a static Go binary that maps itself and
  # signals its own threads for goroutine preemption.
  /usr/local/bin/cloudflared rmix,
  signal (send,receive) peer=@{profile_name},
  @{PROC}/@{pid}/** r,
  @{PROC}/sys/net/core/** r,
  /sys/kernel/mm/transparent_hugepage/hpage_pmd_size r,
  /root/.cloudflared/** rw,

  # QUIC and HTTP/2 to Cloudflare's edge, and the origins it proxies to
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,
  network netlink raw,
}
